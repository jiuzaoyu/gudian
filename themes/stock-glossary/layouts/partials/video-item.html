{{- $v := .video -}} {{- $ctx := .page -}} {{- if not $v }}{{ return }}{{ end -}} {{- $type := $v.type | default "" -}} {{- if not $type -}} {{- if or $v.provider $v.url $v.embed_url -}} {{- $type = "embed" -}} {{- else if $v.src -}} {{- $type = "self" -}} {{- end -}} {{- end -}} {{- $hasEmbed := or $v.url $v.embed_url (and $v.id $v.provider) -}} {{- $src := $v.src | default "" -}} {{- if and (not $src) $v.file -}} {{- $src = printf "/videos/%s" $v.file -}} {{- end -}} {{- if and $src (not (hasPrefix $src "http")) (not (hasPrefix $src "/")) -}} {{- $src = printf "/videos/%s" $src -}} {{- end -}} {{- $cdn := site.Params.video_cdn | default "" -}} {{- if and $src $cdn (not (hasPrefix $src "http")) -}} {{- $src = printf "%s%s" ($cdn | strings.TrimSuffix "/") ($src) -}} {{- end -}} {{- $show := false -}} {{- if eq $type "embed" -}} {{- $show = $hasEmbed -}} {{- else if eq $type "self" -}} {{- $show = ne $src "" -}} {{- else -}} {{- $show = or $hasEmbed (ne $src "") -}} {{- end -}} {{- if not $show }}{{ return }}{{ end -}}
